

================================================================================
TABLE: academic_years
================================================================================
CREATE TABLE `academic_years` (
  `id` bigint unsigned NOT NULL AUTO_INCREMENT,
  `name` varchar(30) NOT NULL,
  `start_date` date NOT NULL,
  `end_date` date NOT NULL,
  `is_current` tinyint(1) NOT NULL DEFAULT '0',
  PRIMARY KEY (`id`),
  UNIQUE KEY `uq_academic_years_name` (`name`)
) ENGINE=InnoDB AUTO_INCREMENT=2 DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci;


================================================================================
TABLE: attendance_records
================================================================================
CREATE TABLE `attendance_records` (
  `session_id` bigint unsigned NOT NULL,
  `course_id` bigint unsigned NOT NULL,
  `academic_year_id` bigint unsigned NOT NULL,
  `student_id` bigint unsigned NOT NULL,
  `status` enum('present','absent','late','excused') NOT NULL,
  `notes` varchar(500) DEFAULT NULL,
  PRIMARY KEY (`session_id`,`student_id`),
  KEY `ix_attendance_student` (`student_id`,`session_id`),
  KEY `fk_attendance_records_session` (`session_id`,`course_id`,`academic_year_id`),
  KEY `fk_attendance_records_enrollment` (`student_id`,`course_id`,`academic_year_id`),
  CONSTRAINT `fk_attendance_records_enrollment` FOREIGN KEY (`student_id`, `course_id`, `academic_year_id`) REFERENCES `student_courses` (`student_id`, `course_id`, `academic_year_id`) ON DELETE RESTRICT,
  CONSTRAINT `fk_attendance_records_session` FOREIGN KEY (`session_id`, `course_id`, `academic_year_id`) REFERENCES `attendance_sessions` (`id`, `course_id`, `academic_year_id`) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci;

================================================================================
TABLE: salary_payments
================================================================================
CREATE TABLE `salary_payments` (
  `id` bigint unsigned NOT NULL AUTO_INCREMENT,
  `staff_id` bigint unsigned NOT NULL,
  `pay_period` date NOT NULL,
  `gross_amount` decimal(12,2) NOT NULL,
  `allowances` decimal(12,2) NOT NULL DEFAULT '0.00',
  `deductions` decimal(12,2) NOT NULL DEFAULT '0.00',
  `net_amount` decimal(12,2) NOT NULL,
  `status` enum('pending','paid','cancelled') NOT NULL DEFAULT 'pending',
  `paid_at` timestamp NULL DEFAULT NULL,
  `processed_by` bigint unsigned DEFAULT NULL,
  `notes` varchar(500) DEFAULT NULL,
  `created_at` timestamp NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`),
  UNIQUE KEY `uq_salary_staff_period` (`staff_id`,`pay_period`),
  KEY `ix_salary_period_status` (`pay_period`,`status`),
  KEY `fk_salary_payments_processor` (`processed_by`),
  CONSTRAINT `fk_salary_payments_processor` FOREIGN KEY (`processed_by`) REFERENCES `users` (`id`) ON DELETE SET NULL,
  CONSTRAINT `fk_salary_payments_staff` FOREIGN KEY (`staff_id`) REFERENCES `staff` (`id`) ON DELETE RESTRICT,
  CONSTRAINT `chk_salary_allowances` CHECK ((`allowances` >= 0)),
  CONSTRAINT `chk_salary_amounts` CHECK ((`net_amount` = ((`gross_amount` + `allowances`) - `deductions`))),
  CONSTRAINT `chk_salary_deductions` CHECK ((`deductions` >= 0)),
  CONSTRAINT `chk_salary_gross_amount` CHECK ((`gross_amount` >= 0)),
  CONSTRAINT `chk_salary_net_amount` CHECK ((`net_amount` >= 0))
) ENGINE=InnoDB AUTO_INCREMENT=6 DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci;
